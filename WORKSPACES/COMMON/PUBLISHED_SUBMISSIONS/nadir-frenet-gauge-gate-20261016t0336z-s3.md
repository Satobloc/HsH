"""Nadir PLAYGROUND/SANDBOX: gauge-invariant Frenet gate in R^4.

The right-body connection is compared in three cases:

  baseline:    Omega0 = J12 + 2 J34
  naive:       Omega0 + eta*k*cos(k*s) J23
  exact gauge: H.T @ Omega0 @ H + theta'(s) J23,
               H = exp(theta J23), theta = eta*sin(k*s)

The exact gauge is generated by Q_g=Q_0 H and H e1=e1, so it cannot change
the centerline tangent Q e1.  The script computes the first two Euclidean
Frenet curvatures directly from Omega and dOmega/ds, integrates kappa_2 and
kappa_2^2, and checks the analytic naive-connection formulas.

All coordinates and s are dimensionless.  Restoring a centerline scale ell,
the reported curvatures are kappa_1*ell and kappa_2*ell, while
integral(kappa_2^2 ds) is multiplied by 1/ell^2.  No physical mechanism or
SAT/H(s)H validity is asserted.
"""

from __future__ import annotations

import json
import math

import numpy as np
from scipy.integrate import simpson
from scipy.linalg import expm


def J(a: int, b: int) -> np.ndarray:
    """Generator with J_ab e_a=e_b and J_ab e_b=-e_a (zero-based)."""
    out = np.zeros((4, 4))
    out[b, a] = 1.0
    out[a, b] = -1.0
    return out


J12, J23, J34 = J(0, 1), J(1, 2), J(2, 3)
OMEGA0 = J12 + 2.0 * J34
E1 = np.array([1.0, 0.0, 0.0, 0.0])


def connection(s: float, eta: float, k: int, kind: str) -> tuple[np.ndarray, np.ndarray]:
    """Return (Omega, dOmega/ds) for one of the three connections."""
    if kind == "baseline":
        return OMEGA0, np.zeros((4, 4))

    theta = eta * math.sin(k * s)
    theta_p = eta * k * math.cos(k * s)
    theta_pp = -eta * k * k * math.sin(k * s)

    if kind == "naive":
        return OMEGA0 + theta_p * J23, theta_pp * J23

    if kind == "exact_gauge":
        H = expm(theta * J23)
        rotated = H.T @ OMEGA0 @ H
        commutator = OMEGA0 @ J23 - J23 @ OMEGA0
        omega = rotated + theta_p * J23
        omega_p = theta_p * (H.T @ commutator @ H) + theta_pp * J23
        return omega, omega_p

    raise ValueError(kind)


def frenet_curvatures(omega: np.ndarray, omega_p: np.ndarray) -> tuple[float, float]:
    """First two curvatures of the unit-speed curve whose tangent is Q e1.

    Orthogonal Q drops out of every norm and dot product.  With w=Omega e1,
    T'=Qw and T''=Q(Omega w + w'), the Frenet decomposition is

      T'' = -kappa1^2 T + kappa1' N1 + kappa1*kappa2 N2.
    """
    w = omega @ E1
    w_p = omega_p @ E1
    body_tpp = omega @ w + w_p
    kappa1 = float(np.linalg.norm(w))
    if kappa1 == 0.0:
        return 0.0, float("nan")
    n1 = w / kappa1
    kappa1_p = float(np.dot(n1, body_tpp))
    residual = body_tpp + (kappa1 * kappa1) * E1 - kappa1_p * n1
    kappa2 = float(np.linalg.norm(residual) / kappa1)
    return kappa1, kappa2


def run_case(eta: float, k: int, n_grid: int = 20001) -> dict:
    s_grid = np.linspace(0.0, 2.0 * math.pi, n_grid)
    rows = {}
    for kind in ("baseline", "naive", "exact_gauge"):
        k1 = np.empty(n_grid)
        k2 = np.empty(n_grid)
        for i, s in enumerate(s_grid):
            omega, omega_p = connection(float(s), eta, k, kind)
            k1[i], k2[i] = frenet_curvatures(omega, omega_p)
        rows[kind] = {
            "max_abs_kappa1_minus_1": float(np.max(np.abs(k1 - 1.0))),
            "max_kappa2": float(np.max(k2)),
            "integral_abs_kappa2": float(simpson(np.abs(k2), x=s_grid)),
            "integral_kappa2_squared": float(simpson(k2 * k2, x=s_grid)),
        }

    expected_l1 = 4.0 * abs(eta) * k
    expected_l2 = math.pi * eta * eta * k * k
    naive = rows["naive"]
    assert abs(naive["integral_abs_kappa2"] - expected_l1) < 2.0e-7
    assert abs(naive["integral_kappa2_squared"] - expected_l2) < 2.0e-12
    assert rows["exact_gauge"]["max_kappa2"] < 2.0e-13
    assert rows["baseline"]["max_kappa2"] < 2.0e-13

    return {
        "eta": eta,
        "k": k,
        "analytic_naive": {
            "kappa1": 1.0,
            "kappa2_of_s": f"abs({eta*k:.17g}*cos({k}*s))",
            "integral_abs_kappa2": expected_l1,
            "integral_kappa2_squared": expected_l2,
        },
        "numerical": rows,
    }


def main() -> None:
    fixtures = [run_case(eta=0.2, k=k) for k in (1, 2, 3, 7)]
    null = run_case(eta=0.0, k=3)
    out = {
        "status": "GEN/DERIVED PLAYGROUND; no physical validation",
        "domain": "Euclidean R4; dimensionless s in [0,2*pi]",
        "fixtures": fixtures,
        "eta_zero_negative_control": null,
    }
    print(json.dumps(out, indent=2))


if __name__ == "__main__":
    main()
