from sympy.tensor.tensor import TensorIndexType, tensor_indices, TensorHead, riemann_cyclic_replace
from sympy import Derivative as D

# Set up metric and coordinates
coords = sp.symbols('t x y z')
g = sp.MutableDenseNDimArray(sp.symbols('g00:44'), (4, 4))  # g_{mu nu}
g_inv = sp.MutableDenseNDimArray(sp.symbols('g00:44_inv'), (4, 4))  # g^{mu nu}

# Christoffel symbol of the second kind
Gamma = sp.MutableDenseNDimArray([[[0]*4 for _ in range(4)] for _ in range(4)], (4,4,4))

for l in range(4):
    for mu in range(4):
        for nu in range(4):
            term = 0
            for rho in range(4):
                term += g_inv[l, rho] * (
                    D(g[rho, mu], coords[nu]) +
                    D(g[rho, nu], coords[mu]) -
                    D(g[mu, nu], coords[rho])
                )
            Gamma[l, mu, nu] = sp.simplify(0.5 * term)

# Riemann tensor (partial only, no full symbolic contraction for performance)
Riemann = sp.MutableDenseNDimArray([[[[0]*4 for _ in range(4)] for _ in range(4)] for _ in range(4)], (4,4,4,4))

for rho in range(4):
    for sigma in range(4):
        for mu in range(4):
            for nu in range(4):
                Riemann[rho, sigma, mu, nu] = (
                    D(Gamma[rho, nu, sigma], coords[mu]) -
                    D(Gamma[rho, mu, sigma], coords[nu])
                )

Ricci = sp.MutableDenseNDimArray([[0]*4 for _ in range(4)], (4,4))
for mu in range(4):
    for nu in range(4):
        Ricci[mu, nu] = sum(Riemann[rho, mu, rho, nu] for rho in range(4))
