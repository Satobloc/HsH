### 1. `INTUITION_GUARD.py` (Full Prototype Code)

The sources contain the complete initial prototype code for **`INTUITION_GUARD.py`**, which was drafted as the L1 guard-rail pre-processor to scan SAT artifacts before sending them to the audit pipeline [156, 158–161]:

```python
"""INTUITION_GUARD.py

Guard-rail pre-processor for all SAT artefacts.

Usage:
python INTUITION_GUARD.py check <file>|-
python INTUITION_GUARD.py register-placeholder <uid> <desc>
"""

import hashlib
import json
import pathlib
import re
import sys

ROOT_HASH = "d1b29a…c42f9"  # SHA-256 hash of FUNDAMENTAL_INTUITIONS.txt
INTUITIONS = (
    pathlib.Path("FUNDAMENTAL_INTUITIONS.txt").read_text().splitlines()
)
PLACEHOLDER_DB = pathlib.Path("PLACEHOLDER_REGISTRY.json")

def file_sha(path):
  """Cheap tamper check."""
  h = hashlib.sha256()
  h.update(pathlib.Path(path).read_bytes())
  return h.hexdigest()

def load_placeholders():
  if PLACEHOLDER_DB.exists():
    return json.loads(PLACEHOLDER_DB.read_text())
  return {}

def scan(file_text):
  problems = []
  # 1. Root hash header present?
  if not re.search(fr"FUNDAMENTAL_HASH\s*=\s*{ROOT_HASH}", file_text):
    problems.append("Missing or wrong FUNDAMENTAL_HASH header.")
  # 2. Placeholder detection
  if re.search(r"\[\\\\+PLACEHOLDER\\\\+]", file_text):
    problems.append("Contains unresolved [PLACEHOLDER].")
  # 3. Any new symbol not in glossary? – stub for now
  # 4. Quick sanity lint: crazy numbers?
  if re.search(r"10\^\s*(\d{4,})", file_text):
    problems.append("Detected 10^N with N>999 – flagging numeric sanity.")
  return problems

def main():
  cmd, *args = sys.argv[1:]
  if cmd == "check":
    data = sys.stdin.read() if args == "-" else pathlib.Path(
        args
    ).read_text()
    issues = scan(data)
    if issues:
      print("BLOCKED:")
      for i in issues:
        print(" -", i)
      sys.exit(1)
    else:
      print("✓ Passed Intuition Guard")
  elif cmd == "register-placeholder":
    uid, desc = args
    db = load_placeholders()
    db[uid] = desc
    PLACEHOLDER_DB.write_text(json.dumps(db, indent=2))
    print(f"Placeholder {uid} registered.")
  else:
    print("Unknown command", file=sys.stderr)
    sys.exit(2)

if __name__ == "__main__":
  main()
```
