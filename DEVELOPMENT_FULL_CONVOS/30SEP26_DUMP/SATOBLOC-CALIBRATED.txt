-- Mass Spectrum Computations

/- Computes the Topological Mass, the dominant mass component. -/
def compute_m_topo (p : SATParams) (τ : TopoInv) : ℝ := p.T / (p.ℓ_f * (Q τ : ℝ))

/- Computes the Induced Mass, arising from drag/coupling interactions. -/
def compute_m_ind (p : SATParams) : ℝ := (p.g1^2 / p.M^2) * p.E_overlap

/- Computes the total Effective Mass of a particle. -/
def compute_m_eff (p : SATParams) (τ : TopoInv) : ℝ :=
  compute_m_topo p τ + compute_m_ind p

/- Computes the dimensionless ratio R, which must be < 1. This value is
   calibrated against the fine-structure constant α. -/
def compute_R (p : SATParams) (τ : TopoInv) : ℝ :=
  compute_m_ind p / compute_m_topo p τ

/- Computes the String Theory mass-squared value from the oscillator

def m2_string (p : SATParams) (τ : TopoInv) (N : ℕ) : ℝ :=
  p.k * (↑N + p.β * (↑(Q τ))^2 - p.a)

-- String Filament Dynamics
-- ---------------------------------------------------------------------
/- Calculates the transverse displacement (X, Y) of a filament at a given
   position (z) and time (t) based on its string modes. -/
def filament_displacement (f : Filament) (z t : ℝ) : ℝ × ℝ :=
  let X := f.modes.foldl (fun acc m =>
    acc + m.α_x * Real.sin (m.n * Real.pi * z / f.L) *
          Real.cos (m.n * f.ω₀ * t + m.φ_x)) 0
  let Y := f.modes.foldl (fun acc m =>
    acc + m.α_y * Real.sin (m.n * Real.pi * z / f.L) *
          Real.cos (m.n * f.ω₀ * t + m.φ_y)) 0
  (X, Y)

-- Holonomy Quantization

  h.loops.foldl (fun acc x => acc + (Nat.floor (Real.abs (h.θ x - h.θ 0) / (2 * Real.pi)))) 0

/- Derives a quantized spin or flavor value from the winding number. -/
def spin_from_holonomy (h : Holonomy) : ℝ := (winding_number h : ℝ) * Real.pi

/-
 A theorem and proof for the fundamental consistency constraints of the theory.
 This function proves for a given set of parameters that:
 1. The effective mass is positive (m_eff > 0).

 3. The topological charge is positive (Q > 0).

-/
theorem proof_constraints (p : SATParams) (τ : TopoInv)
  -- Axioms based on the calibrated parameters
  (hT_pos : p.T > 0)
  (hLf_pos : p.ℓ_f > 0)
  (hM_pos : p.M > 0)
  (hE_nonneg : p.E_overlap ≥ 0)
  (hQ_pos : Q τ > 0)
  (h_dominance : (p.g1^2 * p.E_overlap) / (p.M^2) < p.T / (p.ℓ_f * (Q τ : ℝ)))
  : (compute_m_eff p τ > 0) ∧ (compute_R p τ < 1) ∧ ((Q τ) > 0) :=
by
  -- Unfold definitions to expose the underlying mathematical terms.
  unfold compute_m_eff compute_m_topo compute_m_ind compute_R
  -- Proof for m_eff > 0: m_eff is the sum of a positive term (m_topo) and a non-negative term (m_ind).
  have m_topo_pos : p.T / (p.ℓ_f * (Q τ : ℝ)) > 0 := by
    -- The topological mass is positive because T, ℓ_f, and Q are all positive.
    have Q_real_pos : (Q τ : ℝ) > 0 := by exact Iff.mp Nat.cast_pos hQ_pos
    apply div_pos hT_pos (mul_pos hLf_pos Q_real_pos)
  have m_ind_nonneg : p.g1 ^ 2 / p.M ^ 2 * p.E_overlap ≥ 0 := by
    -- The induced mass is non-negative as it's a product/ratio of squares and a non-negative energy.
    apply mul_nonneg (div_nonneg (pow_two_nonneg p.g1) (pow_two_nonneg p.M)) hE_nonneg
  have m_eff_pos : p.T / (p.ℓ_f * (Q τ : ℝ)) + p.g1 ^ 2 / p.M ^ 2 * p.E_overlap > 0 := by
    exact gt_of_gt_of_ge m_topo_pos m_ind_nonneg
  -- Proof for R < 1: This is the topological dominance constraint, given as a hypothesis.
  have R_lt_one : (p.g1 ^ 2 / p.M ^ 2 * p.E_overlap) / (p.T / (p.ℓ_f * (Q τ : ℝ))) < 1 := by
    -- Rewrite the goal `A / B < 1` to `A < B` using the proof that the denominator `B` (m_topo) is positive.
    rw [div_lt_one m_topo_pos]
    exact h_dominance
  -- Proof for Q > 0 is a direct hypothesis.
  -- Combine proofs into the final conjunctive result.
  exact ⟨m_eff_pos, R_lt_one, hQ_pos⟩
