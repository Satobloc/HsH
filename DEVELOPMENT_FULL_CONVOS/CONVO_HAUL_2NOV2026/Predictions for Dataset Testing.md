```python
    lambda row: galactic_to_cartesian(row['l'], row['b'], row['Distance_kpc']), axis=1
)

# --- STEP 6: Compute 3D Distances Between Pairs ---
pairs = []
for i in range(len(glitching_df)):
    for j in range(i + 1, len(glitching_df)):
        p1, p2 = glitching_df.iloc[i], glitching_df.iloc[j]
        dist = np.linalg.norm(p1['xyz'] - p2['xyz'])
        pairs.append({
            'Pulsar 1': p1['PSRJ'],
            'Pulsar 2': p2['PSRJ'],
            'Distance 1 (kpc)': p1['Distance_kpc'],
            'Distance 2 (kpc)': p2['Distance_kpc'],
            '3D Separation (kpc)': round(dist, 5)
        })

# --- STEP 7: Output Closest Pairs ---
df_pairs = pd.DataFrame(pairs).sort_values(by='3D Separation (kpc)').reset_index(drop=True)
print("\nTop 10 Closest Glitching Pulsar Pairs:")
display(df_pairs.head(10))

# Optionally save the full list
df_pairs.to_csv("ranked_glitching_pulsar_pairs.csv", index=False)
```
