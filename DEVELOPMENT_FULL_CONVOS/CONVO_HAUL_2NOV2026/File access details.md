# Toy size-robustness experiment:
# Does adding SAT-style patches bias half-light radius recovery under a random magnification field?
import numpy as np, pandas as pd, matplotlib.pyplot as plt
from pathlib import Path

base = Path("/mnt/data/sat_rxj1131")

# --- Helpers ---
def make_ring(n, R, thickness=2.0):
    """Create a thin ring on an n x n grid centered at n/2."""
    y, x = np.indices((n,n))
    cx = cy = (n-1)/2
    r = np.hypot(x-cx, y-cy)
    img = np.exp(-0.5*((r-R)/thickness)**2)
    return img / img.sum()

def add_patches(img, N=5, contrast=3.0, area_frac=0.05, rng=None):
    """Add N gaussian patches with given total area fraction and contrast."""
    rng = np.random.default_rng() if rng is None else rng
    n = img.shape[0]
    base_flux = img.sum()
    # approximate patch sigma from area fraction
    total_area = area_frac * n*n
    sigma_pix = np.sqrt(total_area / (2*np.pi*N))
    y, x = np.indices((n,n))
    patch_sum = np.zeros_like(img)
    for _ in range(N):
        cx = rng.uniform(0, n-1)
        cy = rng.uniform(0, n-1)
        ga = np.exp(-((x-cx)**2 + (y-cy)**2)/(2*sigma_pix**2))
        patch_sum += ga
    patch_sum /= patch_sum.sum() + 1e-12
    # scale patches to desired contrast relative to local background
    patched = img + contrast * patch_sum * img.sum()
    return patched / patched.sum()

def half_light_radius(img):
    """Compute circular half-light radius about center."""
    n = img.shape[0]
    y, x = np.indices((n,n))
    cx = cy = (n-1)/2
    r = np.hypot(x-cx, y-cy)
    # sort pixels by radius
    idx = np.argsort(r.ravel())
    cum = np.cumsum(img.ravel()[idx])
    half = 0.5 * img.sum()
    k = np.searchsorted(cum, half)
    return r.ravel()[idx][k]

def make_magnification_map(n, kappa=0.4, gamma=0.2, rng=None):
    """Proxy magnification field: log-normal random field with adjustable variance.
       Not a physical microlensing map but good for a *toy* bias test."""
    rng = np.random.default_rng() if rng is None else rng
    # generate white noise in Fourier space with 1/f^2 spectrum to mimic caustic structure
    noise = rng.normal(size=(n,n))
    F = np.fft.rfft2(noise)
    ky = np.fft.fftfreq(n)[:,None]
    kx = np.fft.rfftfreq(n)[None,:]
    k = np.sqrt(kx*kx + ky*ky)
    filt = 1.0 / (1.0 + (k*20)**2)  # smooth large structures, add caustic-like small scales
    Ff = F * filt
    field = np.fft.irfft2(Ff, s=(n,n))
    # log-normal magnification; tune variance with (kappa,gamma)
    var = 0.5 + 2.0*(kappa+gamma)  # heuristic
    mag = np.exp(field * np.sqrt(var))
    mag /= mag.mean()
    return mag

def convolve_map(img, mag):
    """Apply magnification and renormalize."""
    out = img * mag
    return out / out.sum()

R_true = 60.0  # pixels (arbitrary scale ~ AU after calibration)
ring = make_ring(n, R_true, thickness=2.5)

rng = np.random.default_rng(0)
mag = make_magnification_map(n, rng=rng)

grid_phi = [0.0, 0.05, 0.1, 0.2]
grid_c   = [1.0, 2.0, 4.0]
grid_N   = [0, 3, 6, 10]

rows = []
for phi in grid_phi:
    for c in grid_c:
        for N in grid_N:
            # construct source
            src = ring.copy()
            if N>0 and phi>0:
                src = add_patches(src, N=N, contrast=c, area_frac=phi, rng=rng)
            # sweep a few random dithers across magnification map (simulate motion)
            hs = []
            for _ in range(12):
