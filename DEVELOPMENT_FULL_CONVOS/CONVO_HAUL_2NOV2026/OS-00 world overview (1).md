import random
import re
from pathlib import Path

def extract_prompt_blocks(text):
    sections = re.split(r"(?:=+\s*=+|-{3,}|\n{2,}|\n?/imagine\s?)", text, flags=re.IGNORECASE)
    blocks = [block.strip() for block in sections if block and block.strip()]

    prompt_blocks = []
    for block in blocks:
        if re.search(r"--\w+|::\d", block):
            prompt_blocks.append(block)
            continue
        if block.count("//") >= 2:
            prompt_blocks.append(block)
            continue
        if len(block) > 40 and re.search(
            r"(velvet|lacquer|arsenic|aspic|tuft|chrome|bodega|midway|mare frigoris)",
            block, re.IGNORECASE
        ):
            prompt_blocks.append(block)
            continue
    return prompt_blocks

def extract_random_prompts(file_paths, n=5, seed=None, tour_mode=False):
    if seed is not None:
        random.seed(seed)
    if isinstance(file_paths, (str, Path)):
        file_paths = [file_paths]

    all_blocks = []
    for path in file_paths:
        try:
            with open(path, "r", encoding="utf-8") as f:
                content = f.read()
                all_blocks.extend(extract_prompt_blocks(content))
        except Exception as e:
            print(f"⚠️ Could not read {path}: {e}")

    if not all_blocks:
        return []

    selected = random.sample(all_blocks, min(n, len(all_blocks)))

        wrapped = []
        for i, block in enumerate(selected, 1):
            stop_name = f"STOP {i:02d} — ROLLER DISPENSER"
            wrapped.append(f"🎡 {stop_name}\n{block}\n")
        return wrapped
    else:
        return selected
